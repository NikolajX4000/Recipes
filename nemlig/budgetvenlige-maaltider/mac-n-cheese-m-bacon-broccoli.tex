\title{Mac ’n’ cheese med bacon og broccoli}
\PrepTime{25}
\CookingTime{15--20}
\CookingTempe{200}
\TypeCooking{Budgetvenlige måltider\index[nemlig]{Budgetvenlig!Mac ’n’ cheese med bacon og broccoli}}
\NbPerson{4}
\Image{0 100 800 900}{images/nemlig/mac-n-cheese-m-bacon-broccoli.png}

\begin{ingredient}
    \begin{main}
        \item 1 Broccoli\index[ingredient]{Broccoli}
        \item 400g Pasta\index[ingredient]{Pasta}
        \item Salt\index[ingredient]{Salt}
        \item 2 pk. Bacon\index[ingredient]{Bacon}
        \item 3 spsk. Smør\index[ingredient]{Smør}
        \item 3 spsk. Mel\index[ingredient]{Mel}
        \item 5dl Mælk\index[ingredient]{Mælk}
        \item 2 poser Revet Ost\index[ingredient]{Ost}
        \item Peber\index[ingredient]{Peber}
        \item 10 spsk. Rasp\index[ingredient]{Rasp}
    \end{main}
\end{ingredient}
\begin{recipe}
    \step{Tænd ovnen på 200 grader varmluft og sæt en bagerist i ovnen.}
    \step{Skyl og skær broccoli i små buketter på et skærebræt.}
    \step{Kog pasta i letsaltet vand i en stor gryde, efter pakkens anvisning, til den er al dente.}
    \step{Kog broccolibuketterne i ca. 2 min. sammen med pastaen i den sidste del af pastaens kogetiden. Si vandet fra og sæt pasta og broccoli til side.}
    \step{Skær bacon i tern på et andet skærebræt.}
    \step{Steg bacon på en tør stegepande, til den er sprød. Sæt bacon til side på en tallerken med køkkenrulle.}
    \step{Smelt smør i en stor gryde på mellem varme.}
    \step{Tilføj mel og pisk, til du får en melbolle. Rist melbollen i et minuts tid under konstant omrøring, så melsmagen fjernes.}
    \step{Pisk mælken i lidt efter lidt, til du får en opbagt sauce. Lad saucen simre ved svag varme i et par minutter.}
    \step{Hæld ost (gem 1/4 af osten til topping) i saucen (se tip!) og lad den smelte under omrøring. Smag saucen til med salt og peber.}
    \step{Vend det gemte ost sammen med rasp i en lille skål.}
    \step{Kom pasta, broccoli og bacon i ostesaucen og rør det hele godt sammen. Hæld det hele i et smurt ovnfast fad.}
    \step{Drys fadet med oste-raspen og bag retten i ovnen i ca. 15--20 min., til toppen er gylden og sprød.}
    \step{Servér sprød og cremet mac’n’cheese i fad.}
\end{recipe}
\begin{notes}
    \textbf{Tip!} Opskriften giver en mild ostesauce, men er du osteglad, kan du tilsætte flere forskellige oste som revet danbo, cheddar eller parmesan.
\end{notes}
