\title{Brændende kærlighed med et twist}
\PrepTime{35}
\CookingTime{15--20}
\CookingTempe{}
\TypeCooking{Budgetvenlige måltider\index[nemlig]{Budgetvenlig!Brændende kærlighed med et twist}}
\NbPerson{4}
\Image{0 100 800 900}{images/nemlig/braendende-kaerlighed-med-et-twist.png}

\begin{ingredient}
    \begin{main}
        \item 1800g Kartofler\index[ingredient]{Kartofler}
        \item 2 Porrer\index[ingredient]{Porre}
        \item 2 Løg\index[ingredient]{Løg}
        \item 2 Æbler\index[ingredient]{Æble}
        \item 1 potte Purløg\index[ingredient]{Purløg}
        \item 1 pk. Rødbeder\index[ingredient]{Rødbeder}
        \item 2 spsk. Smør\index[ingredient]{Smør}
        \item 2dl Mælk\index[ingredient]{Mælk}
        \item Salt\index[ingredient]{Salt}
        \item Peber\index[ingredient]{Peber}
        \item 2 pk. Bacon\index[ingredient]{Bacon}
        \item Vand\index[ingredient]{Vand}
    \end{main}
\end{ingredient}
\begin{recipe}
    \step{Skrub eller skræl kartofler og skær dem i små tern. Kog dem i vand uden salt, i 15--20 min. til de er helt bløde og møre. Hæld vandet fra og lad dem dampe af i gryden i et par minutter.}
    \step{Rengør porre og snit i tynde skiver (se tip!).}
    \step{Pil løg og skær i tynde skiver på et nyt skærebræt.}
    \step{Skyl æblet og skær det i tynde både (fjern kernehuset).}
    \step{Skyl og hak purløg fint, og skær rødbeder i strimler. Anret i skåle til servering.}
    \step{Mos kartoflerne med en kartoffelmoser, gaffel eller piskeris.}
    \step{Tilsæt smør og mælk, og rør godt rundt. Smag til med salt og peber.}
    \step{Steg bacon på en stegepande til det er sprødt. Lad det dryppe af på en tallerken med køkkenrulle, men hold så meget fedt som muligt blive tilbage i stegepanden.}
    \step{Steg nu løg i et par minutter i baconfedtet, til de er klare.}
    \step{Tilsæt porrer og steg i ca. 5 min. til porrerne er gyldne og faldet godt sammen.}
    \step{Tilsæt æble og steg ved svag varme til æblerne er møre, og tilsæt gerne en sjat vand, så det ikke er for tørt.}
    \step{Vend bacon tilbage på panden og vend godt rundt i blandingen og smag til med salt og peber.}
    \step{Servér kartoffelmosen på tallerkner og kom bacon, løg, porrer og æbleblandingen ovenpå, pynt med rødbeder og hakket purløg.}
\end{recipe}
\begin{notes}
    \textbf{Tip!} Skær top og rod af porren på et skærebræt. Flæk porren på langs hele vejen igennem og snit porren i tynde skiver. Kom skiverne i en stor skål med vand og rens skiverne fri for jord. Skift vandet et par gange og lad til sidst skiverne dryppe af i en si.
\end{notes}
