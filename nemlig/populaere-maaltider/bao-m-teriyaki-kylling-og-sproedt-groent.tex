\title{Bao med teriyaki-kylling og sprødt grønt}
\PrepTime{40}
\CookingTime{10}
\CookingTempe{}
\TypeCooking{Populære måltider\index[nemlig]{Populære!Bao med teriyaki-kylling og sprødt grønt}}
\NbPerson{4}
\Image{0 100 800 900}{images/nemlig/bao-m-teriyaki-kylling-og-sproedt-groent.png}

\begin{ingredient}
    \begin{main}
        \item 4 stk. Kyllingebrystfilet\index[ingredient]{Kyllingebrystfilet}
        \item Madolie\index[ingredient]{Madolie}
        \item 8 spsk. Teriyakiglaze\index[ingredient]{Teriyakiglaze}
        \item 8 Bao\index[ingredient]{Bao}
        \item 1 Spidskål\index[ingredient]{Spidskål}
        \item 1 bundt Forårsløg\index[ingredient]{Forårsløg}
        \item 300g Gulerødder\index[ingredient]{Gulerødder}
        \item 1 Agurk\index[ingredient]{Agurk}
        \item 1 Lime\index[ingredient]{Lime}
        \item 2 spsk. Sesamolie\index[ingredient]{Sesamolie}
        \item Salt\index[ingredient]{Salt}
        \item Peber\index[ingredient]{Peber}
        \item 30g Sesamfrø\index[ingredient]{Sesamfrø}
    \end{main}
\end{ingredient}
\begin{recipe}
    \step{Skær kylling i mundrette tern på et skærebræt.}
    \step{Varm lidt olie på en stegepande og steg kyllingetern, til de er brunet på alle sider.}
    \step{Tilsæt teriyaki-glaze og vend godt sammen med kyllingen. Lad kyllingen marinere i glazen og stege færdig ved lav varme, imens resten forberedes.}
    \step{Find en si frem. Tag et stykke bagepapir og prik rigeligt med huller i bagepapiret med en kødnål, kniv eller lignende. Læg bagepapiret ned i sien.}
    \step{Kog vand op i en gryde med låg. Vær opmærksom på, at sien skal kunne passe ned i gryden uden at ramme vandet.}
    \step{Læg bao ned i sien med bagepapir og kom sien henover det simrende vand i gryden og læg låg på sien. Læg ikke bao ovenpå hinanden, men damp i stedet et par bao ad gangen, indtil alle bao er dampet.}
    \step{Lad dem dampe i ca. 10 min., indtil dine bao er blevet bløde og luftige. Forbered imens grøntsager og salat.}
    \step{Lad dine bao hvile på en tallerken i ca. 2--4 min., hvor de vendes halvvejs, så de ikke bliver fugtige på den ene side.}
    \step{Pluk de yderste blade af spidskålet, skyl det og snit det i tynde strimler på et andet skærebræt - brug evt. et mandolinjern.}
    \step{Skyl forårsløg og snit fint.}
    \step{Skræl gulerødder og fortsæt med at skrælle dem, så du får tynde gulerodsbånd.}
    \step{Skyl agurk og skær i tynde, korte stænger.}
    \step{Skyl og riv limeskallen på den fine side af et rivejern. Skær i både.}
    \step{Kom alle grøntsagerne i en salatskål og vend godt sammen med sesamolie, limeskal, salt og peber. Smag til med limesaft (gem gerne et par limebåde til servering)}
    \step{Rist sesamfrøene på en varm og tør stegepande. Drys de ristede sesamfrø henover teriyaki-kyllingen inden servering.}
    \step{Lad hver medspiser samle sine egne bao med sprødt grønt og teriyaki-kylling. Nyd det resterende grønt som sprød salat til de bløde bao.}
\end{recipe}
\begin{notes}
    \textbf{Tip!} Skal du dampe bao til 4 eller 5 personer, kan du med fordel have gang i to gryder.
\end{notes}
