\title{Spaghetti carbonara med guanciale}
\PrepTime{20}
\CookingTime{25}
\CookingTempe{}
\TypeCooking{Verdens måltider\index[nemlig]{Verden!Spaghetti carbonara med guanciale}}
\NbPerson{2}
\Image{0 100 800 900}{images/nemlig/spaghetti-carbonara-m-guanciale.png}

\begin{ingredient}
    \begin{main}
        \item 200g Spaghetti\index[ingredient]{Spaghetti}
        \item 1 pk. Guanciale\index[ingredient]{Guanciale}
        \item 250g Parmesan\index[ingredient]{Parmesan}
        \item 2 Æg\index[ingredient]{Æg}
        \item Salt\index[ingredient]{Salt}
        \item Peber\index[ingredient]{Peber}
        \item Persille\index[ingredient]{Persille}
    \end{main}
\end{ingredient}
\begin{recipe}
    \step{På en tør stegepande, steges guanciale ved middel varme.}
    \step{Lad fedtet smelte langsomt, til kødet er gyldent og let sprødt.}
    \step{Tag kødet af panden med en ske, uden fedtet og lad det blive sprødt på en tallerken med køkkenrulle. Gem fedtet til senere.}
    \step{Kog spaghetti i letsaltet vand i en stor gryde, efter pakkens anvisninger, til den er al dente.}
    \step{Gem ca. 0,5dl kogevand pr. medspiser og stil til side. Hæld resten af kogevandet fra spaghettien.}
    \step{Lad spaghettien blive i gryden.}
    \step{Pluk og skyl persille. Hak det fint og sæt til side i en skål til servering.}
    \step{Riv parmesan fint på et rivejern. Gem en lille smule parmesan til servering.}
    \step{Pisk æggene sammen i en røreskål med revet parmesan og rigeligt friskkværnet peber.}
    \step{I gryden med den varme kogte spaghetti tilføjes nu æggemassen – gryden må ikke stå på det varme blus, da retten så kan skille.}
    \step{Vend den varme spaghetti sammen med æggemassen under konstant omrøring i gryden i ca. 1--2 min., indtil saucen begynder at tykne. Justér evt. konsistensen med lidt af kogevandet fra spaghettien.}
    \step{Tilføj stegt guanciale og tilsæt efter smag, noget af det smeltede fedt derfra. Vend godt rundt og smag retten til med salt og peber.}
    \step{Servér straks spaghetti carbonara med guanciale og top med persille og resten af parmesanen.}
\end{recipe}
\begin{notes}
    \textbf{Vidste du...} At guanciale er en italiensk inspireret lufttørret kødvare lavet af svinekæbe. Den er krydret med salt og sort peber som giver en rig og intens smag til retten.
\end{notes}
