\title{Kødboller i cremet peanutsauce}
\PrepTime{30}
\CookingTime{15}
\CookingTempe{}
\TypeCooking{Verdens måltider\index[nemlig]{Verden!Kødboller i cremet peanutsauce}}
\NbPerson{4}
\Image{0 100 800 900}{images/nemlig/koedboller-i-cremet-peanutsauce.png}

\begin{ingredient}
    \begin{main}
        \item 2 poser Ris\index[ingredient]{Ris}
        \item Salt\index[ingredient]{Salt}
        \item 2 Løg\index[ingredient]{Løg}
        \item 2 fed Hvidløg\index[ingredient]{Hvidløg}
        \item 1 Rød Peberfrugt\index[ingredient]{Peberfrugt}
        \item 1 Lime (Skal og Saft)\index[ingredient]{Lime}
        \item 2 pk. Hakket Grisekød\index[ingredient]{Hakket Grisekød}
        \item Peber\index[ingredient]{Peber}
        \item 2 Æg\index[ingredient]{Æg}
        \item 10 spsk. Hvedemel\index[ingredient]{Mel}
        \item 1dl Mælk\index[ingredient]{Mælk}
        \item Madolie\index[ingredient]{Olie}
        \item 2 poser Jordnøddesmør\index[ingredient]{Jordnøddesmør}
        \item 4dl Kokosmælk\index[ingredient]{Kokosmælk}
        \item 1dl Vand\index[ingredient]{Vand}
        \item 4 spsk. Østerssauce\index[ingredient]{Østerssauce}
    \end{main}
\end{ingredient}
\begin{recipe}
    \step{Skyl risene grundigt under koldt vand i en si.}
    \step{Kom de skyllede ris i en gryde med rigeligt friskt vand (minimum 4 dele vand for hver del ris) og lidt salt.}
    \step{Kog risene i ca. 12 min.}
    \step{Sluk for blusset, si det overskydende vand fra risene og lad risene hvile i gryden i ca. 10 min. under låg.}
    \step{Pil og hak løg og hvidløg fint på et skærebræt.}
    \step{Skyl peberfrugt og kassér kernehuset. Skær peberfrugt i tynde strimler.}
    \step{Skyl lime, riv skallen og skær i både.}
    \step{Kom hakket kød i en røreskål og rør det sejt med salt.}
    \step{Tilføj halvdelen af det hakkede løg og hvidløg, æg, hvedemel, mælk og peber. Lad gerne farsen hvile i ca. 10--15 min. på køl.}
    \step{Form farsen til små runde kødboller med en våd teske og din håndflade.}
    \step{Varm en stor stegepande op med lidt olie og steg kødbollerne, til de er brunet godt på alle sider. Stil dem til side på en tallerken.}
    \step{Kom lidt olie på den samme stegepande, som kødbollerne stegte på, og steg resten af det hakkede løg og hvidløg, til løgene er bløde.}
    \step{Tilføj jordnøddesmør, kokosmælk, limeskal og vand og lad saucen simre i et par minutter, til jordnøddesmørret er jævnt fordelt. Tilsæt evt. mere vand hvis saucen er for tyk.}
    \step{Smag til med østerssauce, limesaft, salt og peber.}
    \step{Kom kødbollerne og peberfrugter i peanutsaucen og lad begge dele simre med i ca. 5--8 min.}
    \step{Servér kødboller i cremet peanutkokossauce på en bund af ris.}
\end{recipe}
