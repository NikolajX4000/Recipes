\subsubsection{Toffee Cocktail}
\noindent
\begin{minipage}{.49\textwidth}
    \noindent\textbf{\LARGE Toffee cocktail\index{Toffee cocktail}}
\end{minipage}
\hspace{.02\textwidth}
\begin{minipage}{.49\textwidth}
    \raggedleft
    \textbf{\large Dessert}\\
    \sweettartguage{-0.7}\index[drinks]{Sweet!70\%!Toffee cocktail}
\end{minipage}
\vspace{.5em}\hrule\vspace{.5em}
\noindent
\textbf{\large Flavor Profile:} Caramel\index[flavor-profile]{Caramel}, Sweet\index[flavor-profile]{Sweet}, Almond\index[flavor-profile]{Almond}, Creamy\index[flavor-profile]{Creamy}, Chocolate\index[flavor-profile]{Chocolate}.
\vspace{.5em}\hrule\vspace{.5em}
\noindent
\textbf{\large History:} The Toffee cocktail has become an internet favorite. Regularly found on Pinterest, YouTube, and Instagram, this drink is becoming one of the more popular dessert cocktails. The inventor is unknown, but you'll be sure to impress your friends with this rich, sweet, and delicious drink!
\vspace{.5em}\hrule\vspace{.5em}
\noindent
\begin{minipage}[t]{.49\textwidth}
    \noindent
    \textbf{\large Ingredients}
    \begin{itemize}
        \item Amaretto 30ml\index[ingredient]{Amaretto}
        \item Baileys 30ml\index[ingredient]{Baileys}
        \item Kahlua 30ml\index[ingredient]{Kahlua}
        \item Heavy Cream 60ml\index[ingredient]{Heavy Cream}
    \end{itemize}
    \noindent
    \textbf{\large Garnish}
    \begin{itemize}
        \item Caramel\index[ingredient]{Caramel}
        \item Shaved Chocolate\index[ingredient]{Chocolate}
    \end{itemize}
\end{minipage}
\hspace{.01\textwidth}\vrule\hspace{.01\textwidth}
\begin{minipage}[t]{.49\textwidth}
    \noindent
    \textbf{\large Tools/Glassware}
    \begin{itemize}
        \item Shaker
        \item Strainer
        \item Jigger
        \item Martini Glass\index[glass]{Martini Glass}
    \end{itemize}
\end{minipage}
\vspace{.5em}\hrule\vspace{.5em}
\noindent
\begin{minipage}{\textwidth}
    \noindent
    \textbf{\large Instructions}
    \begin{itemize}
        \item Place all ingredients into a shaker.
        \item Add ice and shake for 20 seconds.
        \item Rim glass with caramel and shaved chocolate.
        \item Strain drink into glass.
        \item Garnish with more caramel and shaved chocolate.
    \end{itemize}
\end{minipage}
\vspace{.5em}\hrule\vspace{.5em}
\noindent
\textbf{\large Fun Tip:} Add caramel and chocolate to the inside of the glass for a better presentation.
