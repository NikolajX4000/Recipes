\subsubsection{Chocolate Martini}
\noindent
\begin{minipage}{.49\textwidth}
    \noindent\textbf{\LARGE Chocolate Martini\index{Chocolate Martini}}
\end{minipage}
\hspace{.02\textwidth}
\begin{minipage}{.49\textwidth}
    \raggedleft
    \textbf{\large Dessert}\\
    \sweettartguage{-0.4}\index[drinks]{Sweet!40\%!Chocolate Martini}
\end{minipage}
\vspace{.5em}\hrule\vspace{.5em}
\noindent
\textbf{\large Flavor Profile:} Sweet\index[flavor-profile]{Sweet}, Chocolate\index[flavor-profile]{Chocolate}, Bitter\index[flavor-profile]{Bitter}, Creamy\index[flavor-profile]{Creamy}.
\vspace{.5em}\hrule\vspace{.5em}
\noindent
\textbf{\large History:} As the story goes the Chocolate Martini was created in 1955 by Elizabeth Taylor and Rock Hudson while filming Giant. They both were staying in rented houses across the street from each other and spent much time together outside of filming. When asked what inspired them, they said, "We both love chocolate and we both love martinis, so why not put chocolate liquor in a vodka martini?" The rest, as they say in Hollywood, is a wrap!
\vspace{.5em}\hrule\vspace{.5em}
\noindent
\begin{minipage}[t]{.49\textwidth}
    \noindent
    \textbf{\large Ingredients}
    \begin{itemize}
        \item Vodka 30ml\index[ingredient]{Vodka}
        \item Kahlua 22,5ml\index[ingredient]{Kahlua}
        \item Crème de Cacao 22,5ml\index[ingredient]{Crème de Cacao}
        \item Baileys 22,5ml\index[ingredient]{Baileys}
        \item Heavy Cream 15ml\index[ingredient]{Heavy Cream}
    \end{itemize}
    \noindent
    \textbf{\large Garnish}
    \begin{itemize}
        \item Chocolate Syrup\index[ingredient]{Chocolate Syrup}
        \item Shaved Chocolate\index[ingredient]{Chocolate}
    \end{itemize}
\end{minipage}
\hspace{.01\textwidth}\vrule\hspace{.01\textwidth}
\begin{minipage}[t]{.49\textwidth}
    \noindent
    \textbf{\large Tools/Glassware}
    \begin{itemize}
        \item Shaker
        \item Strainer
        \item Jigger
        \item Martini Glass\index[glass]{Martini Glass}
    \end{itemize}
\end{minipage}
\vspace{.5em}\hrule\vspace{.5em}
\noindent
\begin{minipage}{\textwidth}
    \noindent
    \textbf{\large Instructions}
    \begin{itemize}
        \item Place all ingredients into a shaker.
        \item Add ice and shake for 20 seconds.
        \item Add chocolate to inside of glass.
        \item Double strain into glass.
        \item Garnish with chocolate syrup and shaved chocolate.
    \end{itemize}
\end{minipage}
\vspace{.5em}\hrule\vspace{.5em}
\noindent
\textbf{\large Helpful Tip:} Use a squeeze bottle or bar spoon for running chocolate around inside of glass.
