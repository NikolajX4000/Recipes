\noindent
\textbf{\large Flavor Profile:} Boozy\index[flavor-profile]{Boozy}, Sweet\index[flavor-profile]{Sweet}, Lightly Herbal\index[flavor-profile]{Herbal}, Bitter\index[flavor-profile]{Bitter}.
\vspace{.5em}\hrule\vspace{.5em}
\noindent
\textbf{\large History:} Vieux Carré translates to "Old Square" and is a term used to reference the French Quarters in New Orleans. It was created by Walter Bergeron in 1938 at the famous Carousel bar inside Hotel Monteleone in New Orleans. At the Carousel Bar you sit down and slowly spin around the bartenders like a carousel. It was frequently visited by many famous authors including Ernest Hemingway and Truman Capote.
\vspace{.5em}\hrule\vspace{.5em}
\noindent
\begin{minipage}[t]{.49\textwidth}
    \noindent
    \textbf{\large Ingredients}
    \begin{itemize}
        \item Cognac 30ml\index[ingredient]{Cognac}
        \item Rye Whiskey 30ml\index[ingredient]{Rye Whiskey}
        \item Sweet Vermouth 30ml\index[ingredient]{Sweet Vermouth}
        \item Benedictine 1 bar spoon\index[ingredient]{Benedictine}
        \item Angostura Bitters 2 dashes\index[ingredient]{Angostura Bitters}
        \item Peychaud's Bitters 2 dashes\index[ingredient]{Peychaud's Bitters}
    \end{itemize}
    \noindent
    \textbf{\large Garnish}
    \begin{itemize}
        \item Lemon Twist\index[ingredient]{Lemon Twist} or Cherry\index[ingredient]{Cherry}
    \end{itemize}
\end{minipage}
\hspace{.01\textwidth}\vrule\hspace{.01\textwidth}
\begin{minipage}[t]{.49\textwidth}
    \noindent
    \textbf{\large Tools/Glassware}
    \begin{itemize}
        \item Mixing Glass
        \item Strainer
        \item Bar Spoon
        \item Jigger
        \item Rocks Glass\index[glass]{Rocks Glass}
    \end{itemize}
\end{minipage}
\vspace{.5em}\hrule\vspace{.5em}
\noindent
\begin{minipage}{\textwidth}
    \noindent
    \textbf{\large Instructions}
    \begin{itemize}
        \item Place all ingredients into mixing glass.
        \item Add ice and stir for 30 seconds.
        \item Strain into rocks glass over ice or large cube.
        \item Garnish with lemon twist or cherry.
    \end{itemize}
\end{minipage}
\vspace{.5em}\hrule\vspace{.5em}
\noindent
