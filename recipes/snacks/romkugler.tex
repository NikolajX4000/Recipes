\title{Romkugler}
\PrepTime{5}
\CookingTime{}
\CookingTempe{}
\TypeCooking{Non-bake snack}
\NbPerson{10}
\Image{0 100 800 900}{images/romkugler.jpg}

\begin{ingredient}
    \begin{main}
        \item 1 Hindbærroulade (kan købes færdig)\index[ingredient]{Hindbærroulade}
        \item 1 Mazarintærte (kan købes færdig)\index[ingredient]{Mazarintærte}
        \item 2 spsk. Hindbærmarmelade\index[ingredient]{Hindbærmarmelade}
        \item 2cl Romessens\index[ingredient]{Romessens}
        \item 2 spsk. Kakaopulver\index[ingredient]{Kakaopulver}
        \item Krymmel til pynt\index[ingredient]{Krymmel}
    \end{main}
\end{ingredient}
\begin{recipe}
    \step{Ælt hindbærrouladen og mazarintærten sammen i en skål. Brug fingrene eller en røremaskine.}
    \step{Tilsæt marmelade, romessens og kakaopulver. Ælt det godt sammen og form dejen til romkugler.}
    \step{Kom lidt krymmel på en tallerken og tril romkuglerne deri.}
    \step{Stil i køleskabet 30 minutter--1 time inden servering. De kan også holde sig nogle dage i køleskabet.}
\end{recipe}
