\title{Chocolate Chipless Cookies}
\PrepTime{60}
\CookingTime{11}
\CookingTempe{180}
\TypeCooking{Cookie}
\NbPerson{16}
\Image{0 100 800 900}{images/chocolate_chipless_cookies.jpg}

\begin{ingredient}
    \begin{main}
        \item 170g Usaltet Smør\index[ingredient]{Smør}
        \item 200g Brun Farin\index[ingredient]{Brun Farin}
        \item 50g Sukker\index[ingredient]{Sukker}
        \item 1 Æg\index[ingredient]{Æg}
        \item 1 Æggeblomme\index[ingredient]{Æggeblomme}
        \item 1 spsk. Vaniljeekstrakt\index[ingredient]{Vaniljeekstrakt}
        \item 219g Mel\index[ingredient]{Mel}
        \item 0,75 tsk. Natron\index[ingredient]{Natron}
        \item 1 tsk. Salt\index[ingredient]{Salt}
        \item Chokoladeknapper\index[ingredient]{Chokoladeknapper}
    \end{main}
\end{ingredient}
\begin{recipe}
    \step{Forvarm ovnen til 180 \degree C.}
    \step{Brun smør ved middel varme under konstant omrøring indtil det begynder at skumme, bliver brunt, og afgiver en nøddeagtig aroma. Afkøl det brunede smør.}
    \step{I en stor skål, bland den afkølede brunede smør, brun farin og hvidt sukker. Pisk, indtil det er godt blandet.  Tilsæt æg, æggeblomme og vaniljeekstrakt. Rør det hele godt sammen.}
    \step{I en anden skål blandes mel, salt og natron. Rør halvdelen af de tørre ingredienser i de våde, indtil det hele samler sig. Tilsæt langsomt resten af melet lidt ad gangen, og stop hvis dejen begynder at blive for tør.}
    \step{Sæt cookiedejen på køl i mindst en halv time eller natten over.}
    \step{Form dejen til små kugler og fordel dem på en bageplade med 5cm mellemrum. Bag i 11 minutter eller indtil småkagerne er gyldne i kanten.}
    \step{Afkøl inden servering.}
\end{recipe}
\begin{notes}
    Hvis ønsket, kan man tilsætte chokoladeknapper til dejkuglerne.
\end{notes}
