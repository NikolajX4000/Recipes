\title{Broccolisalat}
\PrepTime{60}
\CookingTime{}
\CookingTempe{}
\TypeCooking{Salat}
\NbPerson{4}
\Image{0 100 800 900}{images/broccolisalat.jpg}

\begin{ingredient}
    \begin{main}
        \item Dressing
        \item Salat
    \end{main}
    \begin{subingredient}{Dressing}
        \item 2,5dl Creme Fraiche\index[ingredient]{Creme Fraiche}
        \item 2 tsk. Rødvinseddike\index[ingredient]{Eddike}
        \item 1,5 spsk. Sukker\index[ingredient]{Sukker}
    \end{subingredient}
    \begin{subingredient}{Salat}
        \item 300g Bacon\index[ingredient]{Bacon}
        \item 800g Broccoli\index[ingredient]{Broccoli}
        \item 1,5dl Rosiner\index[ingredient]{Rosiner}
        \item 1 Rødløg\index[ingredient]{Rødløg}
        \item 1,5dl Solsikkekerner\index[ingredient]{Solsikkekerner}
    \end{subingredient}
\end{ingredient}
\begin{recipe}
    \step{Rør dressingen}
    \substep[Dressing]{Rør creme fraiche med rødvinseddike og sukker. Lad dressingen stå koldt en times tid.}
    \step{Bland salaten}
    \substep[Salat]{Rist baconen sprød og lad den dryppe af på et stykke køkkenrulle eller lignende.}
    \substep{Skyl og bræk broccolien ud i små buketter.}
    \substep{Pil og hak rødløg i små tern.}
    \substep{Bland broccolibuketter, solsikkekerner, rosiner, rødløg og halvdelen af baconen i en skål.}
    \substep{Hæld dressingen over og vend rundt.}
    \substep{Pynt med resten af baconen og server.}
\end{recipe}
