\title{Fladbrød med ost}
\PrepTime{15}
\CookingTime{15}
\CookingTempe{Middel}
\TypeCooking{Fladbrød}
\NbPerson{4}
\Image{0 100 800 900}{images/fladbrod_med_ost.jpg}

\begin{ingredient}
    \begin{main}
        \item 2dl Varmt Vand\index[ingredient]{Vand}
        \item 1 tsk. Salt\index[ingredient]{Salt}
        \item 300g Mel\index[ingredient]{Mel}
        \item 100g Ost (Gouda)\index[ingredient]{Ost}
        \item 100g Smør\index[ingredient]{Smør}
        \item Sesamfrø\index[ingredient]{Sesamfrø}
    \end{main}
\end{ingredient}
\begin{recipe}
    \step{Hæld vandet i en skål og bland saltet i til det er opløst. Rør derefter halvdelen af mellet i.}
    \step{Tilsæt den anden halvdelen af mellet og ælt dejen. Når dejen er æltet, overdæk dejen og lad den hvile i 10 minutter.}
    \step{Riv osten med den store side af rivejernet.}
    \step{Del dejen i 4 dele og rul dem til bolde.}
    \step{Rul dejen ud så den er meget tynd.}
    \step{Smelt smørret og pensel dejen med det smeltede smør.}
    \step{Kom den revne ost oven på den udrullede dej og strø med sesamfrø.}
    \step{Rul dejen som en pandekage og rul rullen sammen som en snegl. Rul derefter sneglen ud til den er 18cm i diameter.}
    \step{Bag fladbrødet ved middel varme på en pande med smør. Pensel toppen af fladbrødet med smør. Vend brødet en gang i mellem indtil det er gyldent.}
\end{recipe}
