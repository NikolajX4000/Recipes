\title{One-pot pasta med tomater og chorizo}
\PrepTime{30}
\CookingTime{}
\CookingTempe{}
\TypeCooking{Gris}
\NbPerson{4}
\Image{0 100 800 900}{images/one-pot_pasta_med_tomater_og_chorizo.jpg}

\begin{ingredient}
    \begin{main}
        \item Frisk Basilikum\index[ingredient]{Basilikum}
        \item 1 terning Grøntsagsbouillon\index[ingredient]{Bouillon}
        \item 200g Cherrytomater\index[ingredient]{Cherrytomater}
        \item 0,5 tsk. Chiliflager\index[ingredient]{Chiliflager}
        \item 100g Chorizo\index[ingredient]{Chorizo}
        \item 1 dåse Hakkede Tomater\index[ingredient]{Hakkede Tomater}
        \item 2 fed Hvidløg\index[ingredient]{Hvidløg}
        \item 1 Løg\index[ingredient]{Løg}
        \item 1 spsk. Olivenolie\index[ingredient]{Olie}
        \item Parmesan\index[ingredient]{Parmesan}
        \item 400g Pasta\index[ingredient]{Pasta}
        \item Peber\index[ingredient]{Peber}
        \item Salt\index[ingredient]{Salt}
        \item 1 tsk. Sukker\index[ingredient]{Sukker}
        \item 7dl Vand\index[ingredient]{Vand}
    \end{main}
\end{ingredient}
\begin{recipe}
    \step{Hak løg og hvidløg og skær chorizoen i skiver.}
    \step{Varm olien i en stor gryde og sauter løg, hvidløg og chorizo til løgene bliver klare.}
    \step{Halver tomaterne og kom dem i gryden sammen med hakkede tomater, chiliflager, sukker, salt, peber, pasta, vand og bouillonterning.}
    \step{Bring gryden i kog og læg låg på.}
    \step{Lad gryden koge i 10--12 minutter eller indtil pastaen er kogt al dente. Rør rundt undervejs.}
    \step{Server med revet parmesan og frisk basilikum.}
\end{recipe}
