\title{Spaghetti alla Carbonara}
\PrepTime{35}
\CookingTime{}
\CookingTempe{}
\TypeCooking{Gris}
\NbPerson{4}
\Image{0 100 800 900}{images/spaghetti_carbonara.jpg}

\begin{ingredient}
    \begin{main}
        \item Pasta
        \item Guanciale
        \item Sauce
        \item Friskkværnet Sort Peber\index[ingredient]{Peber}
    \end{main}
    \begin{subingredient}{Pasta}
        \item 400g Spaghetti\index[ingredient]{Spaghetti}
        \item Salt (til pastavand)\index[ingredient]{Salt}
    \end{subingredient}
    \begin{subingredient}{Guanciale}
        \item 200g Guanciale (eller pancetta hvis guanciale ikke er tilgængelig)\index[ingredient]{Guanciale}
    \end{subingredient}
    \begin{subingredient}{Sauce}
        \item 200g Pecorino Romano Ost, revet\index[ingredient]{Pecorino}
        \item 4 Store Æg\index[ingredient]{Æg}
    \end{subingredient}
\end{ingredient}
\begin{recipe}
    \step{Tilbered pastaen}
    \substep[Pasta]{Bring en stor gryde med saltet vand i kog.}
    \substep{Tilsæt spaghetti og kog efter pakkens anvisninger, indtil de er al dente.}
    \substep{Når pastaen er kogt, gem cirka en kop af pastavandet og dræn resten.}
    \step{Steg guancialen}
    \substep[Guanciale]{Skær guanciale i små tern eller strimler.}
    \substep{Mens pastaen koger, varm en stor stegepande over middel varme.}
    \substep{Tilsæt guanciale og steg, indtil det bliver sprødt og gyldent.}
    \substep{Fjern fra varmen, men behold fedtet i panden.}
    \step{Lav saucen}
    \substep[Sauce]{Riv Pecorino Romano osten.}
    \substep{Separer æggene, og brug kun æggeblommerne (eller brug hele æg for en mere cremet konsistens).}
    \substep{I en skål, pisk æggeblommerne og bland med den revne Pecorino Romano ost og en generøs mængde friskkværnet sort peber. Bland godt for at skabe en tyk pasta.}
    \step{Saml retten}
    \substep[Samling]{Tilsæt den varme pasta til stegepanden med guanciale og bland godt for at belægge pastaen med fedtet.}
    \substep{Fjern stegepanden fra varmen og tilsæt hurtigt ægge- og osteblandingen, mens du kaster energisk for at kombinere. Den resterende varme fra pastaen vil forsigtigt tilberede æggene og skabe en cremet sauce.}
    \substep{Tilsæt lidt af det gemte pastavand efter behov for at opnå den ønskede konsistens.}
    \substep{Server straks, med ekstra Pecorino Romano og sort peber på toppen, hvis det ønskes.}
\end{recipe}
