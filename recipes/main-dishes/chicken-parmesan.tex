\title{Chicken parmesan}
\PrepTime{30}
\CookingTime{}
\CookingTempe{}
\TypeCooking{Kylling}
\NbPerson{4}
\Image{0 100 800 900}{images/chicken_parmesan.jpg}

\begin{ingredient}
    \begin{main}
        \item 1 tsk. Basilikum\index[ingredient]{Basilikum}
        \item Frisk Basilikum\index[ingredient]{Basilikum}
        \item 0,5 tsk. Hvidløgspulver\index[ingredient]{Hvidløgspulver}
        \item 600g Kyllingebryst\index[ingredient]{Kyllingebryst}
        \item 0,5 tsk. Løgpulver\index[ingredient]{Løgpulver}
        \item 125g Frisk Mozzarella\index[ingredient]{Mozzarella}
        \item 1 tsk. Oregano\index[ingredient]{Oregano}
        \item Parmesan, revet\index[ingredient]{Parmesan}
        \item 80g Rasp\index[ingredient]{Rasp}
        \item Smør\index[ingredient]{Smør}
        \item 400g Tomatsauce\index[ingredient]{Tomatsauce}
        \item 2 Æg\index[ingredient]{Æg}
    \end{main}
\end{ingredient}
\begin{recipe}
    \step{Opvarm ovnen til 250 grader.}
    \step{Bank kyllingefileterne flade så de er ca. 1cm tykke og halver dem på det lange led.}
    \step{Pisk ægget i en dyb tallerken.}
    \step{Bland raspen sammen med krydderierne.}
    \step{Vend kyllingefileterne i æg først og derefter i rasp.}
    \step{Varm smør op på en pande og steg kyllingefileterne på begge sider til de er gennemstegt.}
    \step{Fordel tomatsaucen i et ovnfast fad og læg kyllingefileterne ovenpå.}
    \step{Drys med revet parmesan og bræk til sidst den friske mozzarella i stykker og fordel ovenpå.}
    \step{Sæt fadet i ovnen og bag i 5--10 minutter, til osten er gylden og smeltet.}
    \step{Server med frisk basilikum.}
\end{recipe}
