\title{Mexicansk kylling i sovs}
\PrepTime{30}
\CookingTime{60}
\CookingTempe{200}
\TypeCooking{Kylling}
\NbPerson{4}
\Image{0 100 800 900}{images/mexicansk_kylling_i_sovs.jpg}

\begin{ingredient}
    \begin{main}
        \item 4 spsk. Brun Farin\index[ingredient]{Brun Farin}
        \item 0,25 tsk. Chili\index[ingredient]{Chili}
        \item 0,5dl Eddike\index[ingredient]{Eddike}
        \item 3 spsk. Engelsk Sauce\index[ingredient]{Engelsk Sauce}
        \item 1dl Ketchup\index[ingredient]{Ketchup}
        \item 600g Kyllingebryst\index[ingredient]{Kyllingebryst}
        \item 1,5 tsk. Paprika\index[ingredient]{Paprika}
        \item 0,25 tsk. Peber\index[ingredient]{Peber}
        \item Ris\index[ingredient]{Ris}
        \item 1,5 tsk. Sennep\index[ingredient]{Sennep}
        \item 2dl Tomatsauce\index[ingredient]{Tomatsauce}
        \item 1dl Vand\index[ingredient]{Vand}
    \end{main}
\end{ingredient}
\begin{recipe}
    \step{Kom alle ingredienserne, på nær kyllingen, i en gryde og kog i 15 minutter.}
    \step{Kom kyllingen i et ildfast fad og hæld sovsen over.}
    \step{Sæt fadet i ovnen i cirka 1 time ved 200 grader.}
    \step{Serveres f.eks. med ris.}
\end{recipe}
