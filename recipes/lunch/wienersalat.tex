\title{Wienersalat}
\PrepTime{15}
\CookingTime{}
\CookingTempe{}
\TypeCooking{Pålægssalat}
\NbPerson{4}
\Image{0 100 800 900}{images/wienersalat.jpg}

\begin{ingredient}
    \begin{main}
        \item 800g Pølser\index[ingredient]{Pølser}
        \item 200g Mayonnaise\index[ingredient]{Mayonnaise}
        \item 1 spsk. Sennep\index[ingredient]{Sennep}
        \item Salt\index[ingredient]{Salt}
        \item Peber\index[ingredient]{Peber}
        \item Frisk Purløg\index[ingredient]{Purløg}
    \end{main}
\end{ingredient}
\begin{recipe}
    \step{Mayonnaise røres med sennep, salt og peber og smages til.}
    \step{Kogte røde pølser eller ufarvede alt efter smag. Skæres i tynde skiver.}
    \step{Pølserne vendes i dressingen og Wienersalaten er klar til servering, pyntes gerne med friskklippet purløg.}
\end{recipe}
