\title{Cookie brownie}
\PrepTime{50}
\CookingTime{20--25}
\CookingTempe{190}
\TypeCooking{Kage}
\NbPerson{12}
\Image{0 100 800 900}{images/cookie_brownie.jpg}

\begin{ingredient}
    \begin{main}
        \item Browniedej
        \item Cookiedej
        \item 2 pk. Oreo\index[ingredient]{Oreo}
    \end{main}
    \begin{subingredient}{Cookiedej}
        \item 1 tsk. Bagepulver\index[ingredient]{Bagepulver}
        \item 150g Brun Farin\index[ingredient]{Brun Farin}
        \item 150g Mørk Chokolade\index[ingredient]{Chokolade}
        \item 250g Hvedemel\index[ingredient]{Mel}
        \item 0,5 tsk. Natron\index[ingredient]{Natron}
        \item 1 knivspids Salt\index[ingredient]{Salt}
        \item 150g Smør\index[ingredient]{Smør}
        \item 1 Æg\index[ingredient]{Æg}
        \item 1 Æggeblomme\index[ingredient]{Æggeblomme}
    \end{subingredient}
    \begin{subingredient}{Browniedej}
        \item 0,5 tsk. Bagepulver\index[ingredient]{Bagepulver}
        \item 250g Mørk Chokolade\index[ingredient]{Chokolade}
        \item 50g Kakaopulver\index[ingredient]{Kakaopulver}
        \item 80g Hvedemel\index[ingredient]{Mel}
        \item 1 drys Salt\index[ingredient]{Salt}
        \item 120g Smør\index[ingredient]{Smør}
        \item 400g Sukker\index[ingredient]{Sukker}
        \item 5 Æg\index[ingredient]{Æg}
    \end{subingredient}
\end{ingredient}
\begin{recipe}
    \step{Mix cookiedejen}
    \substep[Cookiedej]{Stil smørret ud på køkkenbordet i god tid, så den er blød og stuetempereret.}
    \substep{Pisk smør og brun farin til en cremet lys masse. Det tager ca. 3--5 minutter.}
    \substep{Pisk æg og æggeblomme i dejen.}
    \substep{Bland mel, bagepulver, natron og salt. Sigt blandingen i skålen og rør cookiedejen ensartet.}
    \substep{Hak chokoladen og vend i dejen.}
    \step{Mix browniedejen}
    \substep[Browniedej]{Smelt chokolade og smør i en gryde indtil det er blevet en ensartet masse. Lad det køle helt af efterfølgende.}
    \substep{Pisk æg og sukker sammen til en luftig masse.}
    \substep{Vend chokoladeblandingen forsigtigt sammen med æggemassen.}
    \substep{Tilsæt resten af ingredienserne og vend det forsigtigt sammen, sådan så al luften ikke slås ud af æggemassen. Si gerne de tørre ingredienser ned i, for at undgå klumper.}
    \step{Saml kagen}
    \substep[Samling]{Fordel cookiedejen i bunden af en bageform på ca. 25$\times$30 beklædt med bagepapir.}
    \substep{Placer Oreos ovenpå cookiedejen.}
    \substep{Hæld browniedejen udover til sidst, så alle Oreosne er dækket.}
    \substep{Bag kagen ved 190 grader alm. ovn i ca. 20--25 minutter.}
\end{recipe}
